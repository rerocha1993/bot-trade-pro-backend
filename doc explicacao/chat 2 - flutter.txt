# services/ponto_institucional_service.py

from sqlalchemy.orm import Session
from models.ponto_institucional import PontoInstitucional

class PontoInstitucionalService:
    @staticmethod
    def criar_ponto(db: Session, ponto_data: dict):
        ponto = PontoInstitucional(**ponto_data)
        db.add(ponto)
        db.commit()
        db.refresh(ponto)
        return ponto

    @staticmethod
    def listar_pontos(db: Session, par=None, exchange=None, status=None, tipo=None):
        query = db.query(PontoInstitucional)
        if par:
            query = query.filter(PontoInstitucional.par == par)
        if exchange:
            query = query.filter(PontoInstitucional.exchange == exchange)
        if tipo:
            query = query.filter(PontoInstitucional.tipo == tipo)
        if status == "ativo":
            query = query.filter(PontoInstitucional.ativo == True, PontoInstitucional.rompido == False)
        if status == "testado":
            query = query.filter(PontoInstitucional.testado == True, PontoInstitucional.rompido == False)
        if status == "rompido":
            query = query.filter(PontoInstitucional.rompido == True)
        return query.all()

    @staticmethod
    def atualizar_status(db: Session, ponto_id: int, status: str):
        ponto = db.query(PontoInstitucional).filter(PontoInstitucional.id == ponto_id).first()
        if not ponto:
            return None

        if status == "testar":
            ponto.testado = True
            ponto.data_testado = func.now()
        elif status == "romper":
            ponto.rompido = True
            ponto.ativo = False
            ponto.data_rompido = func.now()

        db.commit()
        db.refresh(ponto)
        return ponto

    @staticmethod
    def historico(db: Session, par=None, exchange=None):
        query = db.query(PontoInstitucional)
        if par:
            query = query.filter(PontoInstitucional.par == par)
        if exchange:
            query = query.filter(PontoInstitucional.exchange == exchange)
        return query.order_by(PontoInstitucional.data_identificacao.desc()).all()

# services/detector_ponto_institucional_service.py

from sqlalchemy.orm import Session
from models.ponto_institucional import PontoInstitucional
from datetime import datetime

class DetectorPontoInstitucionalService:
    @staticmethod
    def registrar_ponto(
        db: Session,
        preco: float,
        tipo: str,
        motivo: str,
        par: str,
        exchange: str,
        relevancia: float = 1.0,
        extra: dict = None
    ):
        # Verifica se já existe ponto igual (ativo e não rompido) nesse par/tipo/preço
        ponto_existente = db.query(PontoInstitucional).filter(
            PontoInstitucional.preco == preco,
            PontoInstitucional.par == par,
            PontoInstitucional.exchange == exchange,
            PontoInstitucional.tipo == tipo,
            PontoInstitucional.rompido == False
        ).first()

        if ponto_existente:
            return ponto_existente

        ponto = PontoInstitucional(
            preco=preco,
            tipo=tipo,
            motivo=motivo,
            ativo=True,
            testado=False,
            rompido=False,
            data_identificacao=datetime.utcnow(),
            par=par,
            exchange=exchange,
            relevancia=relevancia,
            extra=str(extra) if extra else None
        )
        db.add(ponto)
        db.commit()
        db.refresh(ponto)
        return ponto

# services/ponto_institucional_service.py (adicione método)

from sqlalchemy import and_

class PontoInstitucionalService:
    # ... métodos anteriores
    @staticmethod
    def atualizar_pontos_testados_ou_rompidos(db: Session, preco_atual: float, par: str, exchange: str):
        pontos_ativos = db.query(PontoInstitucional).filter(
            PontoInstitucional.par == par,
            PontoInstitucional.exchange == exchange,
            PontoInstitucional.ativo == True,
            PontoInstitucional.rompido == False
        ).all()
        for ponto in pontos_ativos:
            # Considera o ponto testado se tocou no preço, rompido se ultrapassou
            if not ponto.testado and abs(ponto.preco - preco_atual) <= 0.0001:  # tolerância ajustável
                ponto.testado = True
                ponto.data_testado = datetime.utcnow()
            elif not ponto.rompido and (
                (ponto.preco < preco_atual and ponto.tipo in ['TPO', 'HVN']) or
                (ponto.preco > preco_atual and ponto.tipo in ['SUPORTE', 'LQN'])
            ):
                ponto.rompido = True
                ponto.ativo = False
                ponto.data_rompido = datetime.utcnow()
        db.commit()

No mesmo par

Na mesma exchange

No mesmo tipo (Ex: TPO, VWAP, OrderBlock, etc)

Com preço igual (ou dentro de uma tolerância, por exemplo, 0.01%)

Que ainda não foi rompido (ativo)

Se for rompido, marca como rompido, inativo e grava a data.

Se só foi testado (mas não rompido), só marca testado.

Ao final, executa db.commit() para salvar no banco.

from datetime import datetime

class PontoInstitucionalService:
    # ... outros métodos ...

    @staticmethod
    def atualizar_pontos_testados_ou_rompidos(db, preco_atual, par, exchange, tolerancia=0.01):
        pontos_ativos = db.query(PontoInstitucional).filter(
            PontoInstitucional.par == par,
            PontoInstitucional.exchange == exchange,
            PontoInstitucional.ativo == True,
            PontoInstitucional.rompido == False
        ).all()
        for ponto in pontos_ativos:
            preco_ponto = float(ponto.preco)
            preco_atual_f = float(preco_atual)
            status_modificado = False

            # TESTADO: se o preço tocou (ajuste tolerância conforme ativo)
            if not ponto.testado and abs(preco_ponto - preco_atual_f) <= tolerancia:
                ponto.testado = True
                ponto.data_testado = datetime.utcnow()
                status_modificado = True
                # Reduz relevância institucional a cada teste (mas não zerar)
                if ponto.relevancia is not None and ponto.relevancia > 0.2:
                    ponto.relevancia = round(ponto.relevancia - 0.2, 2)
                print(f"[LOG] Ponto TESTADO: {ponto.preco} ({ponto.tipo}) em {ponto.par}/{ponto.exchange} | Relevância: {ponto.relevancia} | {ponto.data_testado}")

            # ROMPIDO: se o preço ultrapassou
            if not ponto.rompido:
                if (
                    # Resistência rompida para cima
                    ponto.tipo.upper().startswith(('TPO', 'HVN', 'VWAP', 'ORDER_BLOCK_BULLISH')) and preco_atual_f > preco_ponto
                    # Suporte rompido para baixo
                    or ponto.tipo.upper().startswith(('SUPORTE', 'LQN', 'ORDER_BLOCK_BEARISH')) and preco_atual_f < preco_ponto
                ):
                    ponto.rompido = True
                    ponto.ativo = False
                    ponto.data_rompido = datetime.utcnow()
                    status_modificado = True
                    print(f"[LOG] Ponto ROMPIDO: {ponto.preco} ({ponto.tipo}) em {ponto.par}/{ponto.exchange} | Data: {ponto.data_rompido}")

            # (Opcional) Log se nada mudou:
            # else:
            #     print(f"[LOG] Sem alteração em {ponto.preco} ({ponto.tipo})")

        db.commit()

class DetectorPontoInstitucionalService:
    @staticmethod
    def registrar_ponto(
        db, preco, tipo, motivo, par, exchange, relevancia=1.0, extra=None
    ):
        # Verifica se já existe ponto igual (ativo e não rompido)
        ponto_existente = db.query(PontoInstitucional).filter(
            PontoInstitucional.preco == preco,
            PontoInstitucional.par == par,
            PontoInstitucional.exchange == exchange,
            PontoInstitucional.tipo == tipo,

            par=par,
            exchange=exchange,
            relevancia=relevancia,
            extra=str(extra) if extra else None

FASE 6 – Integração dos Pontos Institucionais com o Robô Trader
Objetivo
Fazer o robô trader (dentro do StrategyEngine ou do executor principal) consultar os pontos institucionais ativos/testados antes de tomar decisão de trade e adaptar sua lógica para:

Evitar trades contra zonas institucionais relevantes

Usar os pontos para decisão de take-profit, stop-loss ou filtros de operação

Considerar confluências (exemplo: não comprar perto de resistência institucional, não vender em suporte forte, etc.)

Checklist Profissional
Consultar pontos institucionais ativos/testados antes de decidir

Adaptar lógica do robô/engine:

Evita compra perto de resistência institucional ativa/forte

Evita venda perto de suporte institucional ativo/forte

Ajusta TP/SL para regiões institucionais mais próximas

Justificar decisões e logar uso dos pontos institucionais

(Opcional) Passar status e influência dos pontos para o frontend/overlay (mostrar “trade evitado por zona institucional”, etc)

from services.ponto_institucional_service import PontoInstitucionalService

class StrategyEngine:
    def __init__(self, ohlcv, bot, db, ...):  # Adicione db na assinatura!
        self.ohlcv = ohlcv
        self.bot = bot
        self.db = db
        # ... outros params

    def obter_pontos_institucionais_proximos(self, preco_atual, tipo=None, raio=0.15):
        # Busca pontos institucionais ativos/testados próximos ao preço
        query = self.db.query(PontoInstitucional).filter(
            PontoInstitucional.par == self.bot.ativo,
            PontoInstitucional.exchange == self.bot.exchange,
            PontoInstitucional.ativo == True,
            PontoInstitucional.rompido == False,
        )
        if tipo:
            query = query.filter(PontoInstitucional.tipo.like(f"%{tipo.upper()}%"))
        pontos = query.all()
        proximos = [p for p in pontos if abs(float(p.preco) - float(preco_atual)) <= raio]
        return proximos

    def decidir(self):
        preco_atual = self.ohlcv[-1][4]

        # Checa zonas institucionais de resistência (TPO_VAH, HVN, etc)
        resistencias = self.obter_pontos_institucionais_proximos(preco_atual, tipo="VAH")
        suportes = self.obter_pontos_institucionais_proximos(preco_atual, tipo="VAL")

        # Exemplo: Evitar compra se preço próximo de resistência institucional forte
        if resistencias and self.bot.tipo_estrategia in ("scalping", "swing"):
            return {"acao": "hold", "motivo": "Próximo de resistência institucional (VAH/HVN)"}
        # Exemplo: Evitar venda se preço próximo de suporte institucional forte
        if suportes and self.bot.tipo_estrategia in ("scalping", "swing"):
            return {"acao": "hold", "motivo": "Próximo de suporte institucional (VAL/LVN)"}

        # ... segue lógica de decisão normal (votação dos indicadores)
        # Se não está em zona institucional, avalia compra/venda

        # Exemplo de como usar para definir TP/SL
        # Se for comprar, usar resistência mais próxima como TP; se for vender, suporte mais próximo como TP
        tp = None
        sl = None
        if self.bot.tipo_estrategia in ("scalping", "swing"):
            if resistencias:
                tp = min([float(p.preco) for p in resistencias])  # TP curto, segurança institucional
            if suportes:
                sl = min([float(p.preco) for p in suportes])  # SL curto

        # ... lógica dos indicadores e votação aqui

Resumo
Antes de tomar qualquer trade, o robô checa se existe ponto institucional próximo.

Se sim, pode segurar/evitar a entrada (“hold”).

Se não, segue com a lógica normal de indicadores.

TP e SL podem ser ajustados automaticamente para regiões institucionais.

O motivo da decisão passa a informar que está sendo influenciado por zona institucional, para máxima transparência no log e no overlay.

Como integrar
Adapte o StrategyEngine para receber o objeto db da sessão ao ser criado.

Use métodos semelhantes a obter_pontos_institucionais_proximos para suporte e resistência, passando o tipo de ponto conforme a lógica desejada.

O raio/tolerância (ex: 0.15%) pode ser ajustado conforme o ativo ou volatilidade.
