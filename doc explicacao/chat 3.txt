def executar_bot_individual(bot_id):
    session = SessionLocal()
    bot = session.query(Bot).filter(Bot.id == bot_id).first()
    if bot:
        executor = BotExecutor(bot, db=session)
        executor.executar()

def main():
    session = SessionLocal()
    bots_ativos = session.query(Bot).filter(Bot.ativo_bot == True).all()

    processos = []
    for bot in bots_ativos:
        p = Process(target=executar_bot_individual, args=(bot.id,))
        p.start()
        processos.append(p)

    for p in processos:
        p.join()

if __name__ == "__main__":
    main()
