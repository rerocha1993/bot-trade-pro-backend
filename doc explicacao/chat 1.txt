# Arquivo: trading_bot/core/fibonacci.py

class Fibonacci:
    def __init__(self, high: float, low: float):
        self.high = high
        self.low = low
        self.levels = self.calcular_niveis()

    def calcular_niveis(self):
        diff = self.high - self.low
        return {
            '0.0%': round(self.low, 2),
            '23.6%': round(self.low + 0.236 * diff, 2),
            '38.2%': round(self.low + 0.382 * diff, 2),
            '50.0%': round(self.low + 0.5 * diff, 2),
            '61.8%': round(self.low + 0.618 * diff, 2),
            '100.0%': round(self.high, 2),
            '161.8%': round(self.low + 1.618 * diff, 2)
        }

    def exibir_niveis(self):
        print("Níveis de Fibonacci:")
        for nivel, valor in self.levels.items():
            print(f"{nivel}: {valor}")

# Arquivo: trading_bot/core/tpo.py

import pandas as pd

class TPOCalculator:
    def __init__(self, precos: list[float], intervalo: float = 50):
        """
        precos: lista de preços (float) geralmente obtidos de fechamento.
        intervalo: tamanho dos bins de agrupamento (ex: 50 = agrupa por cada 50 unidades de preço)
        """
        self.intervalo = intervalo
        self.df = pd.DataFrame(precos, columns=['price'])
        self.df['price_bin'] = (self.df['price'] // self.intervalo) * self.intervalo
        self.tpo = self.df.groupby('price_bin').size().reset_index(name='frequencia')

    def calcular_poc_e_valores(self):
        poc_row = self.tpo.loc[self.tpo['frequencia'].idxmax()]
        valor_medio = self.tpo['frequencia'].mean()

        value_area = self.tpo[self.tpo['frequencia'] >= valor_medio]

        resultado = {
            'POC': poc_row['price_bin'],
            'VAL': value_area['price_bin'].min(),
            'VAH': value_area['price_bin'].max()
        }

        return resultado

    def exibir_tpo(self):
        print(self.tpo)

# Exemplo de uso:
if __name__ == "__main__":
    from trading_bot.core.conexao_bybit import BybitConnector

    tpo = TPOCalculator(precos, intervalo=50)
    tpo.exibir_tpo()

    resultado = tpo.calcular_poc_e_valores()
    print("POC (Point of Control):", resultado['POC'])
    print("Value Area Low (VAL):", resultado['VAL'])
    print("Value Area High (VAH):", resultado['VAH'])

# Arquivo: trading_bot/core/order_flow.py

class OrderFlowAnalyzer:
    def __init__(self, order_book: dict, profundidade: int = 10):
        """
        order_book: dicionário com 'bids' e 'asks', conforme retornado pela Bybit.
        profundidade: número de níveis a considerar no cálculo.
        """
        self.bids = order_book['bids'][:profundidade]
        self.asks = order_book['asks'][:profundidade]

    def calcular_desequilibrio(self):
        total_bids = sum([qtd for preco, qtd in self.bids])
        total_asks = sum([qtd for preco, qtd in self.asks])
        return total_bids - total_asks

    def interpretar(self):
        desequilibrio = self.calcular_desequilibrio()
        if desequilibrio > 0:
            return "Bullish (mais demanda que oferta)"
        elif desequilibrio < 0:
            return "Bearish (mais oferta que demanda)"
        else:
            return "Neutro (equilíbrio entre bids e asks)"

# Exemplo de uso:

    flow = OrderFlowAnalyzer(order_book)
    print("Desequilíbrio:", flow.calcular_desequilibrio())
    print("Interpretação:", flow.interpretar())

# Arquivo: trading_bot/core/order_block.py

class OrderBlockDetector:
    def __init__(self, ohlcv: list):
        """
        ohlcv: lista de candles no formato [timestamp, open, high, low, close, volume]
        """
        self.ohlcv = ohlcv

    def identificar_blocks(self, tipo='bullish'):
        """
        tipo: 'bullish' ou 'bearish'
        Retorna uma lista com os candles identificados como blocos institucionais.
        """
        blocks = []

        for i in range(len(self.ohlcv) - 2):
            atual = self.ohlcv[i]
            prox = self.ohlcv[i + 1]

            open_atual, close_atual = atual[1], atual[4]
            open_prox, close_prox = prox[1], prox[4]

            if tipo == 'bullish':
                # Último candle vermelho seguido por verde forte
                if close_atual < open_atual and close_prox > open_prox and close_prox > close_atual:
                    blocks.append({
                        'index': i,
                        'tipo': 'bullish',
                        'zona': (atual[1], atual[2])  # open e high do candle vermelho
                    })

            elif tipo == 'bearish':
                # Último candle verde seguido por vermelho forte
                if close_atual > open_atual and close_prox < open_prox and close_prox < close_atual:
                    blocks.append({
                        'index': i,
                        'tipo': 'bearish',
                        'zona': (atual[1], atual[3])  # open e low do candle verde

# Arquivo: trading_bot/core/vwap.py

import pandas as pd

class VWAPCalculator:
    def __init__(self, ohlcv: list):
        """
        ohlcv: lista de candles no formato [timestamp, open, high, low, close, volume]
        """
        self.df = pd.DataFrame(ohlcv, columns=["timestamp", "open", "high", "low", "close", "volume"])
        self.df['tp'] = (self.df['high'] + self.df['low'] + self.df['close']) / 3  # preço típico
        self.df['tpv'] = self.df['tp'] * self.df['volume']
        self.df['cum_tpv'] = self.df['tpv'].cumsum()
        self.df['cum_volume'] = self.df['volume'].cumsum()
        self.df['vwap'] = self.df['cum_tpv'] / self.df['cum_volume']

    def obter_vwap_atual(self):
        return round(self.df['vwap'].iloc[-1], 2)

    def exibir_vwap_series(self):
        return self.df[['timestamp', 'vwap']]

class StrategyEngine:
    def __init__(self, sinais: dict, regras: dict = None):
        """
        sinais: dicionário com sinais dos indicadores. Ex:
        {
            "elliott": {"onda3_valida": True},
            "fibonacci": {"suporte_atingido": True},
            "tpo": {"poc_acima": True},
            "order_flow": "Bullish",
            "vwap": {"preco_acima_vwap": True}
        }

        regras: opcional, dicionário que permite personalizar a decisão.
        """
        self.sinais = sinais
        self.regras = regras or {
            "min_votos": 2  # mínimo de sinais positivos além do Elliott
        }

    def decidir(self):
        if not self.sinais.get("elliott", {}).get("onda3_valida"):
            return {"acao": "hold", "motivo": "Elliott não validado"}

        votos = 0
        if self.sinais.get("fibonacci", {}).get("suporte_atingido"):
            votos += 1
        if self.sinais.get("tpo", {}).get("poc_acima"):
            votos += 1
        if self.sinais.get("order_flow") == "Bullish":
            votos += 1
        if self.sinais.get("vwap", {}).get("preco_acima_vwap"):
            votos += 1

        if votos >= self.regras["min_votos"]:
            return {"acao": "buy", "motivo": f"{votos} indicadores a favor"}
        elif votos <= -self.regras["min_votos"]:
            return {"acao": "sell", "motivo": f"{abs(votos)} indicadores contra"}
        else:
            return {"acao": "hold", "motivo": "Votos insuficientes"}

# Exemplo de uso:

        "vwap": {"preco_acima_vwap": False}
    }

    engine = StrategyEngine(sinais_exemplo)

            writer.writeheader()
            writer.writerows(self.resultados)
        print("Resultados salvos em:", path)

from trading_bot.core.conexao_bybit import BybitConnector

class ExecutorReal:
    def __init__(self, par, modo_teste=False):
        self.par = par
        self.bybit = BybitConnector().exchange
        self.modo_teste = modo_teste

    def executar_ordem(self, direcao, quantidade, preco_entrada=None, stop_loss=None, take_profit=None):
        """
        direcao: 'buy' ou 'sell'
        quantidade: tamanho da posição
        preco_entrada: preço limite (opcional)
        stop_loss / take_profit: preços de SL/TP (opcional)
        """
        if self.modo_teste:
            print("[MODO TESTE] Ordem simulada:", direcao, quantidade, self.par)
            return

        tipo_ordem = 'limit' if preco_entrada else 'market'
        params = {}

        if tipo_ordem == 'limit':
            params['price'] = preco_entrada

        order = self.bybit.create_order(
            symbol=self.par,
            type=tipo_ordem,
            side=direcao,
            amount=quantidade,
            params=params
        )
        print("Ordem executada:", order)

        # Stop Loss
        if stop_loss:
            sl_order = self.bybit.create_order(
                symbol=self.par,
                type='stop_market',
                side='sell' if direcao == 'buy' else 'buy',
                amount=quantidade,
                params={'stopPrice': stop_loss}
            )
            print("Stop Loss ativado:", sl_order)

        # Take Profit
        if take_profit:
            tp_order = self.bybit.create_order(
                symbol=self.par,
                type='take_profit_market',
                side='sell' if direcao == 'buy' else 'buy',
                amount=quantidade,
                params={'stopPrice': take_profit}
            )
            print("Take Profit ativado:", tp_order)

# Arquivo: trading_bot/core/risk_manager.py

class RiskManager:
    def __init__(self, config):
        """
        config: dicionário com parâmetros de risco, ex:
        {
            "max_trades_dia": 5,
            "max_perda_dia": 100.0,
            "sl_tipo": "fibonacci",  # ou 'volatilidade' ou 'candle'
            "sl_variacao_pct": 1.0
        }
        """
        self.config = config
        self.trades_realizados = 0
        self.perda_acumulada = 0.0

    def pode_operar(self):
        if self.trades_realizados >= self.config["max_trades_dia"]:
            return False, "Limite de trades diário atingido"
        if self.perda_acumulada >= self.config["max_perda_dia"]:
            return False, "Limite de perda diária atingido"
        return True, ""

    def registrar_trade(self, resultado):
        self.trades_realizados += 1
        if resultado < 0:
            self.perda_acumulada += abs(resultado)

    def calcular_stop_loss(self, preco_entrada, dados=None, fib=None, candle_anterior=None):
        tipo = self.config.get("sl_tipo")

        if tipo == "fibonacci" and fib:
            return fib.levels.get("61.8%")

        elif tipo == "volatilidade" and dados:
            ultimos = [c[4] for c in dados[-5:]]  # fechamento
            media = sum(ultimos) / len(ultimos)
            desvio = max(ultimos) - min(ultimos)
            return preco_entrada - desvio

        elif tipo == "candle" and candle_anterior:
            return candle_anterior[3]  # low do candle anterior

        else:  # fallback percentual
            variacao = preco_entrada * (self.config["sl_variacao_pct"] / 100)
            return round(preco_entrada - variacao, 2)

# Arquivo: run_all_bots.py

from multiprocessing import Process
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
from models.bot import Bot, Base
import urllib
import time

# Conexão com o banco
params = urllib.parse.quote_plus(
    "DRIVER={ODBC Driver 17 for SQL Server};"
    "SERVER=(localdb)\\MSSQLLocalDB;"
    "DATABASE=dbo_trading_bot;"
    "Trusted_Connection=yes"
)
engine = create_engine(f"mssql+pyodbc:///?odbc_connect={params}")
SessionLocal = sessionmaker(bind=engine)

def executar_bot(bot: Bot):
    print(f"[{bot.nome}] Inicializando bot com ativo {bot.ativo} no timeframe {bot.timeframe}")
    while True:
        # Aqui você pode colocar as etapas reais:
        # 1. Coletar dados (via conexão_bybit)
        # 2. Calcular indicadores (Elliott, TPO, etc)
        # 3. Decidir com strategy_engine
        # 4. Simular ou executar real (modo_execucao)
        print(f"[{bot.nome}] Executando lógica do bot...")

        # Simulação de espera entre ciclos (ex: 60s)
        time.sleep(60)

def main():
    session = SessionLocal()
    bots_ativos = session.query(Bot).filter(Bot.ativo_bot == True).all()

    processos = []
    for bot in bots_ativos:
        p = Process(target=executar_bot, args=(bot,))
        p.start()
        processos.append(p)

    # Aguardar todos os bots (opcional)
    for p in processos:
        p.join()

if __name__ == "__main__":
    main()
