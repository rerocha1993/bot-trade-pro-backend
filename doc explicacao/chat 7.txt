Backtesting é o processo de simular execuções do bot no passado para verificar se a estratégia realmente daria lucro com dados reais de mercado.

import os
import requests
from datetime import datetime, timedelta
from zipfile import ZipFile

symbol = "BTCUSDT"
interval = "1h"
start_date = datetime(2023, 5, 1)
end_date = datetime(2023, 5, 31)
output_dir = "binance_csv"

os.makedirs(output_dir, exist_ok=True)

current_date = start_date
while current_date <= end_date:
    date_str = current_date.strftime("%Y-%m-%d")
    file_name = f"{symbol}-{interval}-{date_str}.csv.zip"
    url = f"https://data.binance.vision/data/spot/daily/klines/{symbol}/{interval}/{file_name}"
    local_zip = os.path.join(output_dir, file_name)

    try:
        print(f"🔽 Baixando {file_name}...")
        r = requests.get(url, timeout=15)
        if r.status_code == 200:
            with open(local_zip, "wb") as f:
                f.write(r.content)
            with ZipFile(local_zip, 'r') as zip_ref:
                zip_ref.extractall(output_dir)
            os.remove(local_zip)  # opcional: remove o zip depois de extrair
        else:
            print(f"⚠️ Não encontrado: {url}")
    except Exception as e:
        print(f"❌ Erro ao baixar {file_name}: {e}")
    
    current_date += timedelta(days=1)

print("✅ Download completo.")

# core/binance_csv_normalizer.py

import os
import pandas as pd

class BinanceCSVNormalizer:
    def __init__(self, pasta_csv_bruto, arquivo_saida):
        self.pasta_csv_bruto = pasta_csv_bruto
        self.arquivo_saida = arquivo_saida
        self.colunas_binance = [
            "open_time", "open", "high", "low", "close", "volume",
            "close_time", "quote_asset_volume", "number_of_trades",
            "taker_buy_base_vol", "taker_buy_quote_vol", "ignore"
        ]

    def normalizar_e_unificar(self):
        arquivos = [f for f in os.listdir(self.pasta_csv_bruto) if f.endswith(".csv")]
        todos = []

        for arquivo in sorted(arquivos):
            caminho = os.path.join(self.pasta_csv_bruto, arquivo)
            df = pd.read_csv(caminho, header=None, names=self.colunas_binance)

            df["timestamp"] = df["open_time"] // 1000
            df = df[["timestamp", "open", "high", "low", "close", "volume"]]
            todos.append(df)

        df_final = pd.concat(todos, ignore_index=True)
        df_final.to_csv(self.arquivo_saida, index=False)
        return df_final

# core/backtest_engine.py

import pandas as pd

class BacktestEngine:
    def __init__(self, arquivo_csv, estrategia_func=None):
        self.arquivo_csv = arquivo_csv
        self.dados = pd.read_csv(arquivo_csv)
        self.resultados = []
        self.estrategia_func = estrategia_func or self.estrategia_simples

    def estrategia_simples(self, candle_atual, candle_anterior):
        # Exemplo: compra se fechar acima do anterior, vende se fechar abaixo
        if candle_atual["close"] > candle_anterior["close"]:
            return "buy"
        elif candle_atual["close"] < candle_anterior["close"]:
            return "sell"
        else:
            return "hold"

    def executar(self):
        for i in range(1, len(self.dados)):
            anterior = self.dados.iloc[i - 1]
            atual = self.dados.iloc[i]

            acao = self.estrategia_func(atual, anterior)
            resultado = {
                "timestamp": atual["timestamp"],
                "acao": acao,
                "preco": atual["close"]
            }
            self.resultados.append(resultado)

        return pd.DataFrame(self.resultados)

    def salvar_resultado(self, caminho_saida):
        df_resultado = pd.DataFrame(self.resultados)
        df_resultado.to_csv(caminho_saida, index=False)

# core/binance_csv_normalizer.py

import os
import pandas as pd

class BinanceCSVNormalizer:
    def __init__(self, pasta_csv_bruto, arquivo_saida):
        self.pasta_csv_bruto = pasta_csv_bruto
        self.arquivo_saida = arquivo_saida
        self.colunas_binance = [
            "open_time", "open", "high", "low", "close", "volume",
            "close_time", "quote_asset_volume", "number_of_trades",
            "taker_buy_base_vol", "taker_buy_quote_vol", "ignore"
        ]

    def normalizar_e_unificar(self):
        arquivos = [f for f in os.listdir(self.pasta_csv_bruto) if f.endswith(".csv")]
        todos = []

        for arquivo in sorted(arquivos):
            caminho = os.path.join(self.pasta_csv_bruto, arquivo)
            try:
                df = pd.read_csv(caminho, header=None, names=self.colunas_binance)

                df["timestamp"] = df["open_time"] // 1000
                df = df[["timestamp", "open", "high", "low", "close", "volume"]]
                todos.append(df)
            except Exception as e:
                print(f"Erro ao processar {arquivo}: {e}")

        df_final = pd.concat(todos, ignore_index=True)
        df_final.to_csv(self.arquivo_saida, index=False)
        print(f"✅ Arquivo unificado salvo em: {self.arquivo_saida}")
        return df_final

  final url = Uri.parse('https://api.binance.com/api/v3/exchangeInfo');
  final response = await http.get(url);

  if (response.statusCode == 200) {
    final data = jsonDecode(response.body);
    final symbols = data['symbols'] as List<dynamic>;

import 'dart:convert';
import 'dart:html' as html;
import 'dart:ui' as ui;
import 'package:flutter/material.dart';
import '../services/api_service.dart';

class ChartViewWidget extends StatefulWidget {
  final String ativo;
  final String timeframe;
  final List<Map<String, dynamic>> sinais;
  final List<Map<String, dynamic>> elliott;
  final List<Map<String, dynamic>> candles;
  final Map<String, dynamic>? tpo;

  const ChartViewWidget({
    super.key,
    required this.ativo,
    required this.timeframe,
    required this.sinais,
    required this.elliott,
    required this.candles,
    this.tpo,
  });

  @override
  State<ChartViewWidget> createState() => _ChartViewWidgetState();
}

class _ChartViewWidgetState extends State<ChartViewWidget> {
  late TextEditingController _ativoController;
  List<String> _ativos = [];
  final String viewId = 'chart-html-view';

  @override
  void initState() {
    super.initState();
    _ativoController = TextEditingController(text: widget.ativo);
    _setupIframe();
    WidgetsBinding.instance.addPostFrameCallback((_) => enviarDadosParaIframe());
    carregarAtivos();
  }

  @override
  void didUpdateWidget(covariant ChartViewWidget oldWidget) {
    super.didUpdateWidget(oldWidget);
    enviarDadosParaIframe();
  }

  Future<void> carregarAtivos() async {
    final lista = await ApiService.getBinanceSymbols();
    setState(() => _ativos = lista);
  }

  void _setupIframe() {
    ui.platformViewRegistry.registerViewFactory(viewId, (int viewId) {
      final iframe = html.IFrameElement()
        ..src = 'assets/tradingview_chart.html'
        ..style.border = 'none'
        ..style.height = '100%'
        ..style.width = '100%';
      return iframe;
    });
  }

  void enviarDadosParaIframe() {
    final candles = jsonEncode(widget.candles);
    final sinais = jsonEncode(widget.sinais);
    final elliott = jsonEncode(widget.elliott);
    final tpo = jsonEncode(widget.tpo ?? {});

    html.window.postMessage('loadCandles:$candles', '*');
    html.window.postMessage('loadSinais:$sinais', '*');
    html.window.postMessage('plotElliott:$elliott', '*');
    html.window.postMessage('plotTPO:$tpo', '*');
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        Autocomplete<String>(
          optionsBuilder: (TextEditingValue textEditingValue) {
            if (textEditingValue.text == '') return const Iterable<String>.empty();
            return _ativos.where((String option) => option.toLowerCase().contains(textEditingValue.text.toLowerCase()));
          },
          fieldViewBuilder: (context, controller, focusNode, onEditingComplete) {
            controller.text = _ativoController.text;
            return TextField(
              controller: controller,
              focusNode: focusNode,
              onEditingComplete: onEditingComplete,
              decoration: const InputDecoration(
                labelText: 'Par de Ativo',
                border: OutlineInputBorder(),
              ),
              onSubmitted: (v) => setState(() => _ativoController.text = v),
            );
          },
          onSelected: (String selection) {
            setState(() => _ativoController.text = selection);
          },
        ),
        const SizedBox(height: 12),
        SizedBox(
          height: 400,
          child: HtmlElementView(viewType: viewId),
        ),
      ],
    );
  }
}
