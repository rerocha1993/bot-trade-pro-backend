# core/telegram_alert_service.py
from telegram import Bot, ParseMode
from telegram.error import TelegramError
from config import TELEGRAM_BOT_TOKEN, TELEGRAM_CHAT_ID
import logging

class TelegramAlertService:
    def __init__(self):
        self.bot = Bot(token=TELEGRAM_BOT_TOKEN)
        self.chat_id = TELEGRAM_CHAT_ID

    def enviar_alerta(self, mensagem: str):
        try:
            self.bot.send_message(
                chat_id=self.chat_id,
                text=mensagem,
                parse_mode=ParseMode.HTML
            )
            logging.info("✅ Alerta enviado com sucesso via Telegram.")
        except TelegramError as e:
            logging.error(f"🚫 Falha no envio de alerta via Telegram: {e}")

    def alerta_ordem(self, bot_nome, par, acao, preco):
        mensagem = f"🚨 <b>Ordem executada</b>\n\nBot: {bot_nome}\nPar: {par}\nAção: {acao.upper()}\nPreço: {preco:.2f}"
        self.enviar_alerta(mensagem)

    def alerta_erro(self, bot_nome, erro):
        mensagem = f"❗ <b>Erro detectado</b>\n\nBot: {bot_nome}\nErro: {erro}"
        self.enviar_alerta(mensagem)

    def alerta_status(self, mensagem_status):
        mensagem = f"ℹ️ <b>Status Atualizado</b>\n\n{mensagem_status}"
        self.enviar_alerta(mensagem)

# services/bot_executor.py
from core.telegram_alert_service import TelegramAlertService
from core.conexao_bybit import BybitConnector
from core.strategy_engine import StrategyEngine
from simulador.simulador import SimuladorTrade
from core.executor_real import ExecutorReal
from models.log_execucao import LogExecucaoBot
from datetime import datetime
import logging

class BotExecutor:
    def __init__(self, bot, db):
        self.bot = bot
        self.db = db
        self.alertas = TelegramAlertService()
        self.con = BybitConnector()
        self.executor = ExecutorReal(par=self.bot.ativo, modo_teste=(self.bot.modo_execucao == 'simulador'))
        self.sim = SimuladorTrade(self.bot.nome)

    def executar(self):
        while self.bot.ativo_bot:
            try:
                ohlcv = self.con.obter_ohlcv(self.bot.ativo, timeframe=self.bot.timeframe, limite=200)
                preco_atual = ohlcv[-1][4]

                engine_decisao = StrategyEngine(ohlcv)
                decisao = engine_decisao.decidir()

                acao = decisao["acao"]
                motivo = decisao["motivo"]

                if acao == "buy":
                    self.executor.executar_ordem("buy", quantidade=self.bot.quantidade)
                    self.alertas.alerta_ordem(self.bot.nome, self.bot.ativo, "buy", preco_atual)

                elif acao == "sell":
                    self.executor.executar_ordem("sell", quantidade=self.bot.quantidade)
                    self.alertas.alerta_ordem(self.bot.nome, self.bot.ativo, "sell", preco_atual)

                # Gravar o log
                log = LogExecucaoBot(
                    bot_id=self.bot.id,
                    data_execucao=datetime.utcnow(),
                    preco=preco_atual,
                    acao=acao,
                    motivo=motivo,
                    resultado=None,
                    erro=None
                )
                self.db.add(log)
                self.db.commit()

                logging.info(f"Bot {self.bot.nome} executou: {acao} por {motivo}")

            except Exception as e:
                self.alertas.alerta_erro(self.bot.nome, str(e))
                log = LogExecucaoBot(
                    bot_id=self.bot.id,
                    data_execucao=datetime.utcnow(),
                    preco=0,
                    acao="hold",
                    motivo="erro",
                    resultado=None,
                    erro=str(e)
                )
                self.db.add(log)
                self.db.commit()

                logging.error(f"Erro no bot {self.bot.nome}: {e}")

# services/bot_executor.py

import logging
from datetime import datetime
import asyncio
from core.telegram_alert_service import TelegramAlertService
from core.conexao_bybit import BybitConnector
from core.strategy_engine import StrategyEngine
from simulador.simulador import SimuladorTrade
from core.executor_real import ExecutorReal
from models.log_execucao import LogExecucaoBot

class BotExecutor:
    def __init__(self, bot, db):
        self.bot = bot
        self.db = db
        self.alertas = TelegramAlertService()
        self.con = BybitConnector()
        self.executor = ExecutorReal(par=self.bot.ativo, modo_teste=(self.bot.modo_execucao == 'simulador'))
        self.sim = SimuladorTrade(self.bot.nome)

    def executar(self):
        while self.bot.ativo_bot:
            try:
                ohlcv = self.con.obter_ohlcv(self.bot.ativo, timeframe=self.bot.timeframe, limite=200)
                preco_atual = ohlcv[-1][4]

                engine_decisao = StrategyEngine(ohlcv)
                decisao = engine_decisao.decidir()

                acao = decisao["acao"]
                motivo = decisao["motivo"]

                if acao == "buy":
                    self.executor.executar_ordem("buy", quantidade=self.bot.quantidade)
                    asyncio.run(self.alertas.alerta_ordem(self.bot.nome, self.bot.ativo, "buy", preco_atual))

                elif acao == "sell":
                    self.executor.executar_ordem("sell", quantidade=self.bot.quantidade)
                    asyncio.run(self.alertas.alerta_ordem(self.bot.nome, self.bot.ativo, "sell", preco_atual))

                # Gravar log da execução
                log = LogExecucaoBot(
                    bot_id=self.bot.id,
                    data_execucao=datetime.utcnow(),
                    preco=preco_atual,
                    acao=acao,
                    motivo=motivo,
                    resultado=None,
                    erro=None
                )
                self.db.add(log)
                self.db.commit()

                logging.info(f"✅ {self.bot.nome} executou: {acao.upper()} – {motivo}")

            except Exception as e:
                logging.error(f"❌ Erro no bot {self.bot.nome}: {e}")
                try:
                    asyncio.run(self.alertas.alerta_erro(self.bot.nome, str(e)))
                except Exception as erro_alerta:
                    logging.error(f"❌ Falha ao alertar erro via Telegram: {erro_alerta}")

                log = LogExecucaoBot(
                    bot_id=self.bot.id,
                    data_execucao=datetime.utcnow(),
                    preco=0,
                    acao="hold",
                    motivo="erro",
                    resultado=None,
                    erro=str(e)
                )
                self.db.add(log)
                self.db.commit()

# services/bot_executor.py

import logging
from datetime import datetime
from core.telegram_alert_service import TelegramAlertService
from core.conexao_bybit import BybitConnector
from core.strategy_engine import StrategyEngine
from simulador.simulador import SimuladorTrade
from core.executor_real import ExecutorReal
from models.log_execucao import LogExecucaoBot

class BotExecutor:
    def __init__(self, bot, db):
        self.bot = bot
        self.db = db
        self.alertas = TelegramAlertService()
        self.con = BybitConnector()
        self.executor = ExecutorReal(
            par=self.bot.ativo, 
            modo_teste=(self.bot.modo_execucao == 'simulador')
        )
        self.sim = SimuladorTrade(self.bot.nome)

    def executar(self):
        while self.bot.ativo_bot:
            try:
                ohlcv = self.con.obter_ohlcv(
                    self.bot.ativo, timeframe=self.bot.timeframe, limite=200
                )

                # Assegure que ohlcv esteja no formato correto (lista de listas)
                preco_atual = ohlcv[-1][4]

                engine_decisao = StrategyEngine(ohlcv)
                decisao = engine_decisao.decidir()

                acao = decisao["acao"]
                motivo = decisao["motivo"]

                if acao == "buy":
                    self.executor.executar_ordem("buy", quantidade=self.bot.quantidade)
                    self.alertas.alerta_ordem(self.bot.nome, self.bot.ativo, "buy", preco_atual)

                elif acao == "sell":
                    self.executor.executar_ordem("sell", quantidade=self.bot.quantidade)
                    self.alertas.alerta_ordem(self.bot.nome, self.bot.ativo, "sell", preco_atual)

                # Log execução
                log = LogExecucaoBot(
                    bot_id=self.bot.id,
                    data_execucao=datetime.utcnow(),
                    preco=preco_atual,
                    acao=acao,
                    motivo=motivo,
                    resultado=None,
                    erro=None
                )
                self.db.add(log)
                self.db.commit()

                logging.info(f"✅ {self.bot.nome} executou: {acao.upper()} – {motivo}")

            except Exception as e:
                erro_str = str(e)
                logging.error(f"❌ Erro no bot {self.bot.nome}: {erro_str}")

                # Tenta enviar alerta via Telegram
                try:
                    self.alertas.alerta_erro(self.bot.nome, erro_str)
                except Exception as erro_alerta:
                    logging.error(f"❌ Falha ao alertar via Telegram: {erro_alerta}")

                # Log do erro
                log = LogExecucaoBot(
                    bot_id=self.bot.id,
                    data_execucao=datetime.utcnow(),
                    preco=0,
                    acao="hold",
                    motivo="erro",
                    resultado=None,
                    erro=erro_str
                )
                self.db.add(log)
                self.db.commit()

# services/telegram_alert_service.py

import asyncio
import logging
from telegram import Bot
from telegram.constants import ParseMode
from config import TELEGRAM_BOT_TOKEN, TELEGRAM_CHAT_ID

class TelegramAlertService:
    def __init__(self):
        self.bot = Bot(token=TELEGRAM_BOT_TOKEN)
        self.chat_id = TELEGRAM_CHAT_ID
        self.loop = asyncio.new_event_loop()
        asyncio.set_event_loop(self.loop)

    def enviar_alerta(self, mensagem: str):
        try:
            self.loop.run_until_complete(
                self.bot.send_message(
                    chat_id=self.chat_id,
                    text=mensagem,
                    parse_mode=ParseMode.HTML
                )
            )
            logging.info("✅ Alerta enviado via Telegram.")
        except Exception as e:
            logging.error(f"🚫 Falha ao enviar alerta via Telegram: {e}")

    def alerta_erro(self, bot_nome, erro):
        mensagem = f"❗ <b>Erro no Bot</b>\nBot: <b>{bot_nome}</b>\nErro: <code>{erro}</code>"
        self.enviar_alerta(mensagem)

    def alerta_ordem(self, bot_nome, par, acao, preco):
        mensagem = (
            f"📈 <b>Ordem Executada</b>\n"
            f"Bot: <b>{bot_nome}</b>\n"
            f"Par: <code>{par}</code>\n"
            f"Ação: <b>{acao.upper()}</b>\n"
            f"Preço: <code>{preco:.2f}</code>"
        )
        self.enviar_alerta(mensagem)

        self.bot = bot

        )
        self.sim = SimuladorTrade(self.bot.nome)

    def executar(self):
        while self.bot.ativo_bot:
            try:
                ohlcv = self.con.obter_ohlcv(
                    self.bot.ativo, timeframe=self.bot.timeframe, limite=200
                )

                # Corrigir acesso ao close do candle
                if not ohlcv or not isinstance(ohlcv[-1], list):
                    raise Exception("Formato de OHLCV inválido.")

                preco_atual = ohlcv[-1][4]

                engine = StrategyEngine(ohlcv)
                decisao = engine.decidir()
                acao = decisao["acao"]
                motivo = decisao["motivo"]

                if acao == "buy":
                    self.executor.executar_ordem("buy", quantidade=self.bot.quantidade)
                    self.alertas.alerta_ordem(self.bot.nome, self.bot.ativo, "buy", preco_atual)

                elif acao == "sell":
                    self.executor.executar_ordem("sell", quantidade=self.bot.quantidade)
                    self.alertas.alerta_ordem(self.bot.nome, self.bot.ativo, "sell", preco_atual)

                log = LogExecucaoBot(
                    bot_id=self.bot.id,
                    data_execucao=datetime.utcnow(),
                    preco=preco_atual,
                    acao=acao,
                    motivo=motivo,
                    resultado=None,
                    erro=None
                )
                self.db.add(log)
                self.db.commit()

            except Exception as e:
                erro_str = str(e)
                logging.error(f"❌ Erro no bot {self.bot.nome}: {erro_str}")
                try:
                    self.alertas.alerta_erro(self.bot.nome, erro_str)
                except Exception as erro_alerta:
                    logging.error(f"❌ Falha ao enviar alerta de erro via Telegram: {erro_alerta}")

                log = LogExecucaoBot(
                    bot_id=self.bot.id,
                    data_execucao=datetime.utcnow(),
                    preco=0,
                    acao="hold",
                    motivo="erro",
                    resultado=None,
                    erro=erro_str
                )
                self.db.add(log)
                self.db.commit()

class BybitConnector:
    def __init__(self, category='linear'):  # 'linear' para perp/future, 'spot' se fosse o caso

        })

    def obter_preco_atual(self, par):
        ticker = self.exchange.fetch_ticker(par)
