# api/routes/log_chart_routes.py

from fastapi import APIRouter
from sqlalchemy.orm import Session
from models import Base
from models.log_execucao import LogExecucaoBot
from api.database import get_db
from fastapi import Depends
from datetime import datetime
import time

router = APIRouter()

@router.get("/logs/{bot_id}/grafico-dados")
def obter_dados_grafico(bot_id: int, limite: int = 100, db: Session = Depends(get_db)):
    logs = (
        db.query(LogExecucaoBot)
        .filter(LogExecucaoBot.bot_id == bot_id)
        .order_by(LogExecucaoBot.data_execucao.desc())
        .limit(limite)
        .all()
    )

    candles = []
    acoes = []

    for log in reversed(logs):  # do mais antigo para o mais novo
        timestamp = int(time.mktime(log.data_execucao.timetuple()))
        preco = log.preco or 0

        # Para simplificação, OHLC todos iguais (em produção, ideal usar dados reais)
        candles.append({
            "time": timestamp,
            "open": preco,
            "high": preco,
            "low": preco,
            "close": preco,
        })

        if log.acao in ["buy", "sell"]:
            acoes.append({
                "tipo": log.acao,
                "preco": preco,
                "timestamp": timestamp
            })

    return {
        "candles": candles,
        "acoes": acoes
    }
