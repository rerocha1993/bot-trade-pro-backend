def evento_importante_agora(pais="united states", relevancia="alta", usar_api=False):
    if usar_api:
        return evento_via_api(pais=pais, relevancia=relevancia)
    else:

def evento_via_scraping(pais="united states", relevancia="alta"):
    try:
        response = requests.get(INVESTING_URL, headers=HEADERS)
        soup = BeautifulSoup(response.text, "html.parser")
        linhas = soup.select("tr.js-event-item")
        agora = datetime.utcnow()

        for linha in linhas:
            pais_elem = linha.select_one(".flagCur.noCur")
            impacto_elem = linha.select_one(".sentiment")
            hora_elem = linha.select_one(".first.left.time")

            if not pais_elem or not impacto_elem or not hora_elem:
                continue

            pais_nome = pais_elem.get("title", "").strip()
            impacto = impacto_elem.get("title", "").strip()
            hora_str = hora_elem.text.strip()

            if pais_nome != PAISES.get(pais.lower()):
                continue

            if impacto != IMPACTOS.get(relevancia.lower()):
                continue

            hora_evento = datetime.strptime(hora_str, "%H:%M")
            hora_evento = agora.replace(hour=hora_evento.hour, minute=hora_evento.minute, second=0, microsecond=0)

            if abs((hora_evento - agora).total_seconds()) <= 300:
                return True

        return False
    except Exception as e:
        print(f"[macro_event_checker] Erro no scraping: {e}")
        return False

def evento_via_api(pais="united states", relevancia="alta"):
    try:

        if not API_KEY:

        url = f"https://api.tradingeconomics.com/calendar/country/{pais.replace(' ', '%20')}?c={API_KEY}"

💬 Enviando Mensagens Automáticas
Para que seu bot envie mensagens automáticas, você pode utilizar a API do Telegram. Aqui está um exemplo simples usando Python:

python
Copiar
Editar
import requests

def send_message(token, chat_id, text):
    url = f"https://api.telegram.org/bot{token}/sendMessage"
    payload = {
        'chat_id': chat_id,
        'text': text
    }
    response = requests.post(url, data=payload)
    return response.json()
Parâmetros:

token: O token de API fornecido pelo BotFather.

chat_id: O ID do chat ou usuário para onde a mensagem será enviada.

text: O conteúdo da mensagem.

E é possível que você tenha usado um token no formato embutido no caminho da URL diretamente, que o Telegram também aceita.

✅ Conclusão prática:
Se https://api.telegram.org/bot78800... funcionou pra você — é esse formato que seu código deve usar.

O que vale é o que o Telegram realmente aceita no seu ambiente, não só o que está nos exemplos da doc.
